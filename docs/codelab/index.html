    2.1 Private Channel + Secure Kernel-Timer Pulse (src/platform/qnx/qnx_timer.cpp)

    C++17 · QNX 8.0src/platform/qnx/qnx_timer.cpp
    // Private channel: no external process can ConnectAttach() to inject spoofed ticks.
impl_->chid = ChannelCreate(_NTO_CHF_PRIVATE);
impl_->coid = ConnectAttach(0, 0, impl_->chid, _NTO_SIDE_CHANNEL, 0);

// Read thread priority AFTER set_thread_sched() — NEVER pass SIGEV_PULSE_PRIO_INHERIT!
struct sched_param cur; int cur_policy = SCHED_OTHER; int prio = 10;
if (pthread_getschedparam(pthread_self(), &cur_policy, &cur) == EOK)
  prio = cur.sched_priority;

struct sigevent ev;
SIGEV_PULSE_INIT(&ev, impl_->coid, prio, kPulseTick, 0);
MsgRegisterEvent(&ev, SYSMGR_COID);       // secure event: kernel/procnto only
timer_create(CLOCK_MONOTONIC, &ev, &impl_->timer);
nsec2timespec(&its.it_value, (uint64_t)period_ns);
nsec2timespec(&its.it_interval, (uint64_t)period_ns);
timer_settime(impl_->timer, 0, &its, nullptr);

// Hot loop (0 heap allocations):
struct _pulse p;
MsgReceivePulse(impl_->chid, &p, sizeof p, nullptr);
    QNX 8.0 Priority Pitfall (SIGEV_PULSE_PRIO_INHERIT): In QNX SDP 8.0, SIGEV_PULSE_PRIO_INHERIT makes the receiving thread run at the process's initial priority (typically 10), silently demoting a SCHED_FIFO 40 control thread on every timer tick. Always configure scheduling first, then construct PeriodicTimer with the thread's explicit priority.

    // Driver executes each 1 kHz step at the client's priority AND on the client's pinned CPU cluster:
int chid = ChannelCreate(_NTO_CHF_UNBLOCK | _NTO_CHF_DISCONNECT | _NTO_CHF_INHERIT_RUNMASK);

int robot_io_msg(resmgr_context_t* ctp, io_msg_t* msg, RESMGR_OCB_T* ocb) {
  if (ctp->rcvid == 0) return ENOTSUP;                  // refuse combine messages
  if (msg->i.mgrid != kQnxRobotMgrId) return ENOSYS;    // _IOMGR_PRIVATE_BASE + 0x52
  if (!(ocb->ioflag & _IO_FLAG_WR)) return EPERM;       // _IO_MSG bypasses iofunc_*_verify!
  resmgr_msgget(ctp, &req, sizeof req, 0);
  g_ctx.server->handle(req, sizeof req, rep);
  MsgReply(ctp->rcvid, EOK, &rep, sizeof rep);
  return _RESMGR_NOREPLY;
}

    5.1 Dual-Layer HAM Self-Healing (apps/relmw_ham_setup.cpp + qnx_timer.cpp)
    
      
    
        In QNX OS 8.0, a heartbeat alone triggers nothing — HAM only reacts to registered conditions. We attach two complementary entities so both hard crashes and silent infinite loops self-heal cleanly:
      
    
      
    
        
    
          
    C++17 · libhamapps/relmw_ham_setup.cpp + qnx_timer.cpp
    
          
        
    

    // Layer 1 (apps/relmw_ham_setup.cpp): respawn supervisor & robotd on process death
ham_connect(0);
ham_entity_t* e    = ham_attach(name, 0, -1, line, 0);
ham_condition_t* c = ham_condition(e, CONDDEATH, "death", HREARMAFTERRESTART);
ham_action_restart(c, "restart", line, HREARMAFTERRESTART);

// Layer 2 (qnx_timer.cpp watchdog_attach): turn a silent hang into a death Layer 1 restarts
ham_entity_t* hb   = ham_attach_self("relmw_supervisor_hb", heartbeat_ns, 3, 10, 0);
ham_condition_t* lo = ham_condition(hb, CONDHBEATMISSEDLOW,  "hb_missed_low",  0);
ham_condition_t* hi = ham_condition(hb, CONDHBEATMISSEDHIGH, "hb_missed_high", 0);
ham_action_execute(hi, "kill_hung", "/system/bin/slay -f -9 -m pid <pid>", 0);
